36struct RollerShutterConfig {
37 uint8_t motorUpsideDown = 1; // 0 - not set/not used, 1 - false, 2 - true
38 uint8_t buttonsUpsideDown = 1; // 0 - not set/not used, 1 - false, 2 - true
39 int8_t timeMargin = -1; // -1 default (device specific), 0 - not set/not used
40 // 1 - no margin,
41 // > 1 - 51% of opening/closing time added on extreme
42 // positions - value should be decremented by 1.
43 uint8_t visualizationType = 0; // 0 - default, other values depends on
44 // Cloud and App support
45};
